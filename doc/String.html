def alphabet_sum
    alphabet = ("a".."z").to_a
    sum = 0
    self.downcase.each_char do |alpha|
       if alphabet.include? (alpha)
        numeric_value = alphabet.index(alpha)  + 1
        sum += numeric_value
       end
    end
    sum
end
